// ── 3D Globe ─────────────────────────────────────────────────────────────────

function GlobeInner() {
  const coreRef = useRef<THREE.Mesh>(null);
  const wire1Ref = useRef<THREE.Mesh>(null);
  const wire2Ref = useRef<THREE.Mesh>(null);
  const groupRef = useRef<THREE.Group>(null);

  useFrame((state, delta) => {
    const t = state.clock.getElapsedTime();
    if (coreRef.current) coreRef.current.rotation.y += delta * 0.12;
    if (wire1Ref.current) wire1Ref.current.rotation.y -= delta * 0.07;
    if (wire1Ref.current) wire1Ref.current.rotation.x += delta * 0.03;
    if (wire2Ref.current) wire2Ref.current.rotation.y += delta * 0.05;
    if (wire2Ref.current) wire2Ref.current.rotation.z -= delta * 0.04;
    if (groupRef.current) {
      groupRef.current.rotation.y = t * 0.04;
      groupRef.current.position.y = Math.sin(t * 0.5) * 0.08;
    }
  });

  return (
    <group ref={groupRef}>
      {/* Solid core sphere */}
      <mesh ref={coreRef}>
        <sphereGeometry args={[1.5, 64, 64]} />
        <meshStandardMaterial
          color="#050914"
          emissive="#0B5FFF"
          emissiveIntensity={0.2}
          roughness={0.35}
          metalness={0.7}
        />
      </mesh>

      {/* Outer wireframe ring 1 */}
      <mesh ref={wire1Ref}>
        <sphereGeometry args={[1.56, 28, 28]} />
        <meshBasicMaterial color="#60A5FA" wireframe transparent opacity={0.25} />
      </mesh>

      {/* Outer wireframe ring 2 — tilted for depth */}
      <mesh ref={wire2Ref} rotation={[Math.PI / 4, 0, Math.PI / 6]}>
        <sphereGeometry args={[1.62, 20, 20]} />
        <meshBasicMaterial color="#0B5FFF" wireframe transparent opacity={0.15} />
      </mesh>

      {/* Accent glow lights */}
      <pointLight position={[2, 1.5, 2]} color="#60A5FA" intensity={3} distance={6} />
      <pointLight position={[-2, -1, 1.5]} color="#0B5FFF" intensity={2} distance={5} />
      <pointLight position={[0, 2, -2]} color="#93C5FD" intensity={1.5} distance={4} />
    </group>
  );
}

function Globe() {
  return (
    <div className="relative flex items-center justify-center">
      <div className="relative z-10 h-[320px] w-[320px] sm:h-[400px] sm:w-[400px] md:h-[480px] md:w-[480px]">
        <Canvas camera={{ position: [0, 0, 4.5], fov: 45 }}>
          <ambientLight intensity={0.35} />
          <directionalLight position={[5, 5, 5]} intensity={0.6} />
          <GlobeInner />
        </Canvas>
      </div>
      {/* Glow halo behind globe */}
      <div className="absolute inset-0 rounded-full bg-accent/10 blur-[80px] -z-10 scale-75" />
    </div>
  );
}
